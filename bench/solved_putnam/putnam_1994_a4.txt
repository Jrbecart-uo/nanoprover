import Mathlib

open Filter Topology

/-!
Putnam 1994 A4.  Over ℤ, M is invertible iff det M = ±1.  f(t) = det(A + tB) =
α + βt + γt² (`Matrix.det_fin_two`, numeral matrices via `Matrix.ofNat_apply`), and
f(0..4) ∈ {±1}.  Third differences of a quadratic vanish: f(t+3) = 3f(t+2) − 3f(t+1) + f(t),
which with ±1 values forces f(0) = … = f(4), hence f(5) = ±1 (`omega` on α, β, γ).
-/
theorem putnam_1994_a4
  (A B : Matrix (Fin 2) (Fin 2) ℤ)
  (ABinv : Nonempty (Invertible A) ∧
    Nonempty (Invertible (A + B)) ∧
    Nonempty (Invertible (A + 2 * B)) ∧
    Nonempty (Invertible (A + 3 * B)) ∧
    Nonempty (Invertible (A + 4 * B)))
  : Nonempty (Invertible (A + 5 * B)) := by
  obtain ⟨⟨i0⟩, ⟨i1⟩, ⟨i2⟩, ⟨i3⟩, ⟨i4⟩⟩ := ABinv
  have hdet : ∀ M : Matrix (Fin 2) (Fin 2) ℤ, Invertible M → M.det = 1 ∨ M.det = -1 := by
    intro M hM
    exact Int.isUnit_iff.mp ((Matrix.isUnit_iff_isUnit_det M).mp (isUnit_of_invertible M))
  set α := A.det
  set β := A 0 0 * B 1 1 + B 0 0 * A 1 1 - A 0 1 * B 1 0 - B 0 1 * A 1 0
  set γ := B.det
  have f0 : A.det = α := rfl
  have f1 : (A + B).det = α + β + γ := by
    simp [α, β, γ, Matrix.det_fin_two]; ring
  have f2 : (A + 2 * B).det = α + 2 * β + 4 * γ := by
    simp [α, β, γ, Matrix.det_fin_two, Matrix.mul_apply, Matrix.ofNat_apply]; ring
  have f3 : (A + 3 * B).det = α + 3 * β + 9 * γ := by
    simp [α, β, γ, Matrix.det_fin_two, Matrix.mul_apply, Matrix.ofNat_apply]; ring
  have f4 : (A + 4 * B).det = α + 4 * β + 16 * γ := by
    simp [α, β, γ, Matrix.det_fin_two, Matrix.mul_apply, Matrix.ofNat_apply]; ring
  have f5 : (A + 5 * B).det = α + 5 * β + 25 * γ := by
    simp [α, β, γ, Matrix.det_fin_two, Matrix.mul_apply, Matrix.ofNat_apply]; ring
  have d0 := hdet _ i0
  have d1 := hdet _ i1
  have d2 := hdet _ i2
  have d3 := hdet _ i3
  have d4 := hdet _ i4
  rw [f0] at d0; rw [f1] at d1; rw [f2] at d2; rw [f3] at d3; rw [f4] at d4
  have h5 : IsUnit (A + 5 * B).det := by
    rw [f5, Int.isUnit_iff]; omega
  exact ((Matrix.isUnit_iff_isUnit_det _).mpr h5).nonempty_invertible
