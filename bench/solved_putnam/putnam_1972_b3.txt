theorem putnam_1972_b3
(G : Type*) [Group G] (A B : G)
(hab : A * B * A = B * A^2 * B ∧ A^3 = 1 ∧ (∃ n : ℤ, n > 0 ∧ B^(2*n - 1) = 1))
: B = 1 := by
  obtain ⟨hr, hA3, n, hnpos, hB⟩ := hab
  have hA1 : A * A ^ 2 = 1 := by
    rw [← pow_succ']
    exact hA3
  have h1 : (A * B * A) ^ 2 = A ^ 2 * B * A ^ 2 := by
    have e1 : (A * B * A) ^ 2 = A * (B * A ^ 2 * B) * A := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [← hr] at e1
    rw [e1]
    simp only [pow_succ, pow_zero, one_mul, mul_assoc]
  have h2 : A ^ 2 * B * A ^ 2 = B⁻¹ * A * B := by
    have e3 : B * (A ^ 2 * B * A ^ 2) = B * A ^ 2 * B * A ^ 2 := by
      simp only [mul_assoc]
    rw [← hr] at e3
    have e4 : A * B * A * A ^ 2 = A * B * A ^ 3 := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e4, hA3, mul_one] at e3
    have e5 : A ^ 2 * B * A ^ 2 = B⁻¹ * (B * (A ^ 2 * B * A ^ 2)) := by
      simp only [inv_mul_cancel_left]
    rw [e5, e3]
    simp only [mul_assoc]
  have h3 : (A * B * A) ^ 6 = 1 := by
    have e6 : (A * B * A) ^ 6 = ((A * B * A) ^ 2) ^ 3 := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e6, h1, h2]
    have e7 : (B⁻¹ * A * B) ^ 3 = B⁻¹ * A ^ 3 * B := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc, mul_inv_cancel_left]
    rw [e7, hA3]
    simp only [mul_one, inv_mul_cancel]
  have h4 : (A * B * A) ^ 3 = A ^ 2 * B ^ 2 * A := by
    have e8 : (A * B * A) ^ 3 = (A * B * A) ^ 2 * (A * B * A) := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e8, h1]
    have e9 : A ^ 2 * B * A ^ 2 * (A * B * A) = A ^ 2 * B * A ^ 3 * B * A := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e9, hA3]
    simp only [pow_succ, pow_zero, one_mul, mul_one, mul_assoc]
  have h5 : (A * B * A) ^ 6 = A ^ 2 * B ^ 4 * A := by
    have e10 : (A * B * A) ^ 6 = ((A * B * A) ^ 3) ^ 2 := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e10, h4]
    have e11 : (A ^ 2 * B ^ 2 * A) ^ 2 = A ^ 2 * B ^ 2 * A ^ 3 * B ^ 2 * A := by
      simp only [pow_succ, pow_zero, one_mul, mul_assoc]
    rw [e11, hA3]
    simp only [pow_succ, pow_zero, one_mul, mul_one, mul_assoc]
  have h6 : B ^ 4 = 1 := by
    have e12 : B ^ 4 = (A ^ 2)⁻¹ * (A ^ 2 * B ^ 4 * A) * A⁻¹ := by
      simp only [mul_assoc, inv_mul_cancel_left, mul_inv_cancel, mul_one]
    rw [e12, ← h5, h3, mul_one, ← mul_inv_rev, hA1, inv_one]
  have hB4z : B ^ (4 : ℤ) = 1 := by
    rw [show ((4 : ℤ)) = ((4 : ℕ) : ℤ) by norm_num, zpow_natCast, h6]
  have key : B = (B ^ (2 * n - 1)) ^ (2 * n - 1) * (B ^ (4 : ℤ)) ^ (n - n * n) := by
    rw [← zpow_mul, ← zpow_mul, ← zpow_add]
    have h : (2 * n - 1) * (2 * n - 1) + 4 * (n - n * n) = 1 := by ring
    rw [h, zpow_one]
  rw [hB, hB4z, one_zpow, one_zpow, one_mul] at key
  exact key
