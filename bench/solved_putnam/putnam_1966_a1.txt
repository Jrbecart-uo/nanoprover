import Mathlib

/-!
Putnam 1966 A1.  `f n = ∑_{m ≤ n} ⌊m/2⌋ = ⌊n²/4⌋` (both branches of the `if` are `m / 2`
for `m ≥ 0`), and `⌊(x+y)²/4⌋ - ⌊(x-y)²/4⌋ = xy` since `(x+y)² = (x-y)² + 4xy`.
-/
theorem putnam_1966_a1
(f : ℤ → ℤ)
(hf : f = fun n : ℤ => ∑ m ∈ Finset.Icc 0 n, (if Even m then m / 2 else (m - 1)/2))
: ∀ x y : ℤ, x > 0 ∧ y > 0 ∧ x > y → x * y = f (x + y) - f (x - y) := by
  have hterm : ∀ m : ℤ, 0 ≤ m → (if Even m then m / 2 else (m - 1) / 2) = m / 2 := by
    intro m _
    split_ifs with h
    · rfl
    · rw [Int.even_iff] at h; omega
  have hstep : ∀ n : ℤ, n ^ 2 / 4 + (n + 1) / 2 = (n + 1) ^ 2 / 4 := by
    intro n
    rcases Int.even_or_odd' n with ⟨j, rfl | rfl⟩
    · have e1 : (2 * j) ^ 2 = 4 * j ^ 2 := by ring
      have e2 : (2 * j + 1) ^ 2 = 4 * (j ^ 2 + j) + 1 := by ring
      rw [e1, e2]; omega
    · have e1 : (2 * j + 1) ^ 2 = 4 * (j ^ 2 + j) + 1 := by ring
      have e2 : (2 * j + 1 + 1) ^ 2 = 4 * (j ^ 2 + 2 * j + 1) := by ring
      rw [e1, e2]; omega
  have hclosed : ∀ k : ℕ, f k = (k : ℤ) ^ 2 / 4 := by
    intro k
    induction k with
    | zero => simp [hf]
    | succ k ih =>
      subst hf
      simp only at ih ⊢
      push_cast
      rw [← Finset.insert_Icc_right_eq_Icc_add_one (by positivity), Finset.sum_insert (by simp),
        hterm _ (by positivity), ih, add_comm, hstep]
  rintro x y ⟨hx, hy, hxy⟩
  obtain ⟨a, ha⟩ := Int.eq_ofNat_of_zero_le (by omega : (0 : ℤ) ≤ x + y)
  obtain ⟨b, hb⟩ := Int.eq_ofNat_of_zero_le (by omega : (0 : ℤ) ≤ x - y)
  rw [ha, hb, hclosed, hclosed, ← ha, ← hb]
  have : (x + y) ^ 2 = (x - y) ^ 2 + 4 * (x * y) := by ring
  rw [this]
  omega
