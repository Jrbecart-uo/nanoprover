import Mathlib

open Topology Filter

/-!
Putnam 1992 A1.  `f = 1 - n` iff `f ∘ f = id`, `f (f (n+2) + 2) = n`, `f 0 = 1`.
Backward: `f ∘ f = id` makes `f` injective, so `f (n+2) + 2 = f n`; with `f 0 = 1`,
`f 1 = f (f 0) = 0`, a two-way induction on `ℤ` gives `f n = 1 - n`.
-/
theorem putnam_1992_a1
  (f : ℤ → ℤ) :
  (f = fun n ↦ 1 - n) ↔
  (∀ n : ℤ, f (f n) = n) ∧ (∀ n : ℤ, f (f (n + 2) + 2) = n) ∧ (f 0 = 1) := by
  constructor
  · rintro rfl
    refine ⟨fun n => by ring, fun n => by ring, by ring⟩
  · rintro ⟨h1, h2, h3⟩
    have hinj : Function.Injective f := fun a b hab => by rw [← h1 a, hab, h1]
    have hstep : ∀ n, f (n + 2) + 2 = f n := fun n => hinj (by rw [h2, h1])
    have hf1 : f 1 = 0 := by rw [← h3, h1]
    have key : ∀ n : ℤ, f n = 1 - n ∧ f (n + 1) = -n := by
      intro n
      induction n using Int.induction_on with
      | zero => exact ⟨by simpa using h3, by simpa using hf1⟩
      | succ k ih =>
        refine ⟨ih.2.trans (by ring), ?_⟩
        have := hstep k
        rw [show (k : ℤ) + 1 + 1 = k + 2 by ring]
        linarith [ih.1]
      | pred k ih =>
        refine ⟨?_, by rw [show -(k : ℤ) - 1 + 1 = -k by ring]; linarith [ih.1]⟩
        have := hstep (-(k : ℤ) - 1)
        rw [show -(k : ℤ) - 1 + 2 = -k + 1 by ring] at this
        linarith [ih.2]
    funext n
    exact (key n).1
