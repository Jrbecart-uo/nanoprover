/-!
Putnam 1971 A3.  For a triangle inscribed in a circle of radius R (center C),
AB²·AC²·BC² = 4R²·D², D the cross product (= twice the signed area): a polynomial identity
modulo the three circle equations, closed by `linear_combination` with multipliers found
by a sympy linear-ansatz solve (circle recentred at the origin).  Lattice vertices make D an
integer; distinct vertices make the product > 0, so D ≠ 0 and |D| ≥ 1.  Hence
AB·AC·BC = √(4R²D²) = 2R|D| ≥ 2R.
-/
theorem putnam_1971_a3
    (a b c : ℝ × ℝ)
    (R : ℝ)
    (habclattice : a.1 = round a.1 ∧ a.2 = round a.2 ∧ b.1 = round b.1 ∧ b.2 = round b.2 ∧ c.1 = round c.1 ∧ c.2 = round c.2)
    (habcneq : a ≠ b ∧ a ≠ c ∧ b ≠ c)
    (hR : R > 0)
    (hcircle : ∃ C : ℝ × ℝ,
      √((a.1 - C.1)^2 + (a.2 - C.2)^2) = R ∧
      √((b.1 - C.1)^2 + (b.2 - C.2)^2) = R ∧
      √((c.1 - C.1)^2 + (c.2 - C.2)^2) = R) :
    (√((a.1 - b.1)^2 + (a.2 - b.2)^2)) * (√((a.1 - c.1)^2 + (a.2 - c.2)^2)) * (√((b.1 - c.1)^2 + (b.2 - c.2)^2)) ≥ 2 * R := by

      ((x1 - x2) ^ 2 + (y1 - y2) ^ 2) * ((x1 - x3) ^ 2 + (y1 - y3) ^ 2) * ((x2 - x3) ^ 2 + (y2 - y3) ^ 2)

  have sq : ∀ z : ℝ, 0 ≤ z → √z = R → z - R ^ 2 = 0 := by
    intro z hz h; rw [← h, Real.sq_sqrt hz]; ring
  have hprod := key a.1 a.2 b.1 b.2 c.1 c.2 C.1 C.2 (R ^ 2)
    (sq _ (by positivity) hA) (sq _ (by positivity) hB) (sq _ (by positivity) hC)
  clear key
  set AB := (a.1 - b.1) ^ 2 + (a.2 - b.2) ^ 2 with hAB
  set AC := (a.1 - c.1) ^ 2 + (a.2 - c.2) ^ 2 with hAC
  set BC := (b.1 - c.1) ^ 2 + (b.2 - c.2) ^ 2 with hBC
  set D := (b.1 - a.1) * (c.2 - a.2) - (c.1 - a.1) * (b.2 - a.2) with hD
  -- D is an integer
  set Dz : ℤ := (round b.1 - round a.1) * (round c.2 - round a.2) - (round c.1 - round a.1) * (round b.2 - round a.2)
  have hDz : D = (Dz : ℝ) := by
    rw [hD, ha1, ha2, hb1, hb2, hc1, hc2]; simp only [Dz]; push_cast; ring
  have side0 : ∀ u v : ℝ × ℝ, u ≠ v → 0 < (u.1 - v.1) ^ 2 + (u.2 - v.2) ^ 2 := by
    intro u v huv
    by_contra hle
    push Not at hle
    apply huv
    have h1 : (u.1 - v.1) ^ 2 = 0 := le_antisymm (by linarith [sq_nonneg (u.2 - v.2)]) (sq_nonneg _)

  have pBC := side0 b c hbc

    rw [h0, Int.cast_zero] at hDz
    rw [hDz] at hprod
    have h0' : AB * AC * BC = 0 := by rw [hprod]; ring
