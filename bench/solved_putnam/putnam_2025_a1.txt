import Mathlib

open Finset

/-!
Putnam 2025 A1.  Let g_k = gcd(2m_k+1, 2n_k+1).  Cross-multiplying the recurrence and using that
m_{k+1}, n_{k+1} are coprime gives m_{k+1} g_k = 2m_k+1 and n_{k+1} g_k = 2n_k+1, hence
v_{k+1} g_k = 2 v_k for v_k = |m_k − n_k| (never 0, since m_0 ≠ n_0 propagates).  So
(∏_{j<k} g_j) v_k = 2^k v_0; the product is odd, hence divides v_0.  Every "bad" k has g_k odd and
≠ 1, i.e. g_k ≥ 3, so 3^{#bad below k} ≤ v_0 for all k, and the bad set is finite.
-/

theorem putnam_2025_a1
  (m n : ℕ → ℕ)
  (hm : ∀ k : ℕ, 0 < m k)
  (hn : ∀ k : ℕ, 0 < n k)
  (h_distinct : m 0 ≠ n 0)
  (h_coprime : ∀ k : ℕ, 0 < k → Nat.Coprime (m k) (n k))
  (h_recurrence : ∀ k : ℕ,
    (m (k + 1) : ℚ) / (n (k + 1) : ℚ) = (2 * (m k : ℚ) + 1) / (2 * (n k : ℚ) + 1)) :
  {k : ℕ | ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)}.Finite := by
  set g : ℕ → ℕ := fun k => Nat.gcd (2 * m k + 1) (2 * n k + 1) with hg
  have hcross : ∀ k, m (k + 1) * (2 * n k + 1) = n (k + 1) * (2 * m k + 1) := by
    intro k
    have h := h_recurrence k
    have h1 : (n (k + 1) : ℚ) ≠ 0 := by exact_mod_cast (hn (k + 1)).ne'
    have h2 : (2 * (n k : ℚ) + 1) ≠ 0 := by positivity
    rw [div_eq_div_iff h1 h2] at h
    have h' : ((m (k + 1) * (2 * n k + 1) : ℕ) : ℚ) = ((n (k + 1) * (2 * m k + 1) : ℕ) : ℚ) := by
      push_cast; linarith
    exact_mod_cast h'
  have hstep : ∀ k, m (k + 1) * g k = 2 * m k + 1 ∧ n (k + 1) * g k = 2 * n k + 1 := by
    intro k
    obtain ⟨a, b, hab, ha, hb⟩ := Nat.exists_coprime (2 * m k + 1) (2 * n k + 1)
    have hG : 0 < g k := Nat.gcd_pos_of_pos_left _ (by omega)
    have ha' : 2 * m k + 1 = a * g k := ha
    have hb' : 2 * n k + 1 = b * g k := hb
    have hc := hcross k
    have e : m (k + 1) * b * g k = n (k + 1) * a * g k := by
      rw [mul_assoc, mul_assoc, ← ha', ← hb']; exact hc
    have e2 : m (k + 1) * b = n (k + 1) * a := Nat.eq_of_mul_eq_mul_right hG e
    have hmn := h_coprime (k + 1) (by omega)
    have hma : m (k + 1) = a := by
      apply Nat.dvd_antisymm
      · exact hmn.dvd_of_dvd_mul_left ⟨b, e2.symm⟩
      · exact hab.dvd_of_dvd_mul_right ⟨n (k + 1), by rw [e2]; ring⟩
    have hnb : n (k + 1) = b := by
      apply Nat.dvd_antisymm
      · exact hmn.symm.dvd_of_dvd_mul_left ⟨a, by rw [← e2]⟩
      · exact hab.symm.dvd_of_dvd_mul_right ⟨m (k + 1), by rw [← e2]; ring⟩
    exact ⟨by rw [hma, ha'], by rw [hnb, hb']⟩
  have hodd : ∀ k, Odd (g k) := by
    intro k
    have : Odd (2 * m k + 1) := odd_two_mul_add_one _
    exact this.of_dvd_nat (Nat.gcd_dvd_left _ _)
  have hne : ∀ k, m k ≠ n k := by
    intro k
    induction k with
    | zero => exact h_distinct
    | succ k ih =>
      intro h
      have := hstep k
      rw [h] at this
      omega
  set v : ℕ → ℕ := fun k => Int.natAbs ((m k : ℤ) - n k) with hvdef
  have hv0 : 0 < v 0 := by
    simp only [hvdef, Int.natAbs_pos]
    have := hne 0; omega
  have hv : ∀ k, v (k + 1) * g k = 2 * v k := by
    intro k
    have h1 := hstep k
    have hz : ((m (k + 1) : ℤ) - n (k + 1)) * (g k : ℤ) = 2 * ((m k : ℤ) - n k) := by
      have a1 : ((m (k + 1) * g k : ℕ) : ℤ) = ((2 * m k + 1 : ℕ) : ℤ) := by rw [h1.1]
      have a2 : ((n (k + 1) * g k : ℕ) : ℤ) = ((2 * n k + 1 : ℕ) : ℤ) := by rw [h1.2]
      push_cast at a1 a2
      linear_combination a1 - a2
    have := congrArg Int.natAbs hz
    simp only [Int.natAbs_mul, Int.natAbs_natCast] at this
    simpa [hvdef] using this
  set P : ℕ → ℕ := fun k => ∏ j ∈ range k, g j with hPdef
  have hP : ∀ k, v k * P k = 2 ^ k * v 0 := by
    intro k
    induction k with
    | zero => simp [hPdef]
    | succ k ih =>
      simp only [hPdef, prod_range_succ] at ih ⊢
      calc v (k + 1) * ((∏ j ∈ range k, g j) * g k) = (v (k + 1) * g k) * ∏ j ∈ range k, g j := by ring
        _ = 2 * (v k * ∏ j ∈ range k, g j) := by rw [hv k]; ring
        _ = 2 ^ (k + 1) * v 0 := by rw [ih]; ring
  have hPle : ∀ k, P k ≤ v 0 := by
    intro k
    have hcop : Nat.Coprime (P k) (2 ^ k) := by
      apply Nat.Coprime.pow_right
      exact Nat.Coprime.prod_left (fun j _ => (hodd j).coprime_two_right)
    have hd : P k ∣ 2 ^ k * v 0 := ⟨v k, by rw [← hP k]; ring⟩
    exact Nat.le_of_dvd hv0 (hcop.dvd_of_dvd_mul_left hd)
  have hbad : ∀ j, g j ≠ 1 → 3 ≤ g j := by
    intro j hj
    have ho := hodd j
    have hpos : 0 < g j := Nat.gcd_pos_of_pos_left _ (by omega)
    rcases ho with ⟨t, ht⟩
    omega
  have hcount : ∀ k, 3 ^ ((range k).filter (fun j => g j ≠ 1)).card ≤ v 0 := by
    intro k
    refine le_trans ?_ (hPle k)
    have : ∏ j ∈ range k, (if g j ≠ 1 then 3 else 1) ≤ ∏ j ∈ range k, g j := by
      apply prod_le_prod'
      intro j _
      split_ifs with h
      · exact hbad j h
      · push Not at h; omega
    rw [prod_ite, prod_const, prod_const_one, mul_one] at this
    exact this
  have hS : {k : ℕ | ¬Nat.Coprime (2 * m k + 1) (2 * n k + 1)} = {k | g k ≠ 1} := by
    ext k; simp [hg, Nat.Coprime]
  rw [hS]
  by_contra hinf
  rw [Set.not_finite] at hinf
  obtain ⟨t, ht, hcard⟩ := hinf.exists_subset_card_eq (v 0 + 1)
  set K := t.sup id + 1
  have hsub : t ⊆ (range K).filter (fun j => g j ≠ 1) := by
    intro j hj
    simp only [mem_filter, mem_range]
    refine ⟨?_, ht hj⟩
    have : j ≤ t.sup id := le_sup (f := id) hj
    omega
  have h1 := card_le_card hsub
  have h2 := hcount K
  have h3 : 3 ^ (v 0 + 1) ≤ 3 ^ ((range K).filter (fun j => g j ≠ 1)).card :=
    Nat.pow_le_pow_right (by norm_num) (by omega)
  have h4 : v 0 + 1 < 3 ^ (v 0 + 1) := Nat.lt_pow_self (by norm_num)
  omega
