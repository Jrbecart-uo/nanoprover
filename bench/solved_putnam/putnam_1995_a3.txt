import Mathlib

open Filter Topology Real

/-!
Putnam 1995 A3.  Write `D`, `E` for the numbers with digits `d`, `e`.  "Replacing digit `i`
of `d` by `e i` gives a multiple of 7" reads `D + (eᵢ - dᵢ) 10ⁱ ≡ 0 (mod 7)`.  Summing over
the 9 digits: `9D + (E - D) = 8D + E ≡ D + E ≡ 0`.  With the same for `e, f`:
`(eᵢ - dᵢ) 10ⁱ ≡ -D` and `(fᵢ - eᵢ) 10ⁱ ≡ -E`, so `(fᵢ - dᵢ) 10ⁱ ≡ -(D + E) ≡ 0`, and
`10ⁱ` is a unit mod 7.  (The digit bounds are not needed.)
-/
theorem putnam_1995_a3
(relation : (Fin 9 → ℤ) → (Fin 9 → ℤ) → Prop)
(digits_to_num : (Fin 9 → ℤ) → ℤ)
(hdigits_to_num : digits_to_num = fun dig => ∑ i : Fin 9, (dig i) * 10^i.1)
(hrelation : ∀ d e : (Fin 9 → ℤ), relation d e ↔ (∀ i : Fin 9, d i < 10 ∧ d i ≥ 0 ∧ e i < 10 ∧ e i ≥ 0) ∧ (∀ i : Fin 9, 7 ∣ (digits_to_num (fun j : Fin 9 => if j = i then e j else d j))))
: ∀ d e f : (Fin 9 → ℤ), ((relation d e) ∧ (relation e f)) → (∀ i : Fin 9, 7 ∣ d i - f i) := by
  -- replacing one digit
  have hrep : ∀ (d e : Fin 9 → ℤ) (i : Fin 9),
      digits_to_num (fun j => if j = i then e j else d j) = digits_to_num d + (e i - d i) * 10 ^ i.1 := by
    intro d e i
    rw [hdigits_to_num]
    simp only
    have : ∀ j : Fin 9, (if j = i then e j else d j) * 10 ^ j.1
        = d j * 10 ^ j.1 + (if j = i then (e j - d j) * 10 ^ j.1 else 0) := by
      intro j; split_ifs <;> ring
    rw [Finset.sum_congr rfl fun j _ => this j, Finset.sum_add_distrib, Finset.sum_ite_eq']
    simp
  have z : ∀ x : ℤ, (7 : ℤ) ∣ x ↔ ((x : ZMod 7) = 0) := fun x => by
    rw [show (7 : ℤ) = ((7 : ℕ) : ℤ) by rfl, ZMod.intCast_zmod_eq_zero_iff_dvd]
  -- the relation, read mod 7
  have hmod : ∀ d e : Fin 9 → ℤ, relation d e →
      (∀ i : Fin 9, ((digits_to_num d : ℤ) : ZMod 7) + ((e i : ZMod 7) - d i) * 10 ^ i.1 = 0) ∧
      ((digits_to_num d : ZMod 7) + digits_to_num e = 0) := by
    intro d e hde
    have h := ((hrelation d e).mp hde).2
    have hi : ∀ i : Fin 9, ((digits_to_num d : ℤ) : ZMod 7) + ((e i : ZMod 7) - d i) * 10 ^ i.1 = 0 := by
      intro i
      have := (z _).mp (h i)
      rw [hrep] at this
      push_cast at this
      exact this
    refine ⟨hi, ?_⟩
    have hsum := Finset.sum_eq_zero (s := Finset.univ) (fun i (_ : i ∈ Finset.univ) => hi i)
    rw [Finset.sum_add_distrib, Finset.sum_const, Finset.card_univ, Fintype.card_fin] at hsum
    have hE : ((digits_to_num e : ℤ) : ZMod 7) - digits_to_num d
        = ∑ i : Fin 9, ((e i : ZMod 7) - d i) * 10 ^ i.1 := by
      rw [hdigits_to_num]; push_cast
      rw [← Finset.sum_sub_distrib]
      exact Finset.sum_congr rfl fun i _ => by ring
    rw [← hE, nsmul_eq_mul] at hsum
    have h8 : ((9 : ℕ) : ZMod 7) = 2 := by decide
    rw [h8] at hsum
    linear_combination hsum
  intro d e f ⟨hde, hef⟩ i
  obtain ⟨h1, hDE⟩ := hmod d e hde
  obtain ⟨h2, -⟩ := hmod e f hef
  have hunit : IsUnit ((10 : ZMod 7) ^ i.1) := (IsUnit.of_mul_eq_one (a := (10 : ZMod 7)) 5 (by decide)).pow _
  have key : (((d i : ℤ) : ZMod 7) - f i) * 10 ^ i.1 = 0 := by
    linear_combination -(h1 i) - h2 i + hDE
  rw [z]
  push_cast
  exact (hunit.mul_left_eq_zero).mp key

#print axioms putnam_1995_a3
