open EuclideanGeometry Topology Filter Complex

/-!
Putnam 1965 B2.  In a round-robin tournament `∑ wᵣ² = ∑ lᵣ²` where `lᵣ = n - 1 - wᵣ`.
`∑ lᵣ² - ∑ wᵣ² = n(n-1)² - 2(n-1) ∑ wᵣ`, and `2 ∑ wᵣ = n(n-1)` because every pair `i ≠ j`
has exactly one winner (`won i j + won j i = 1`).
-/
theorem putnam_1965_b2
(n : ℕ)
(hn : n > 1)
(won : Fin n → Fin n → Bool)
(hirrefl : ∀ i : Fin n, won i i = false)
(hantisymm : ∀ i j : Fin n, i ≠ j → won i j = ¬won j i)
(w l : Fin n → ℤ)
(hw : w = fun r : Fin n => ∑ j : Fin n, (if won r j then 1 else 0))
(hl : l = fun r : Fin n => n - 1 - w r)
: ∑ r : Fin n, (w r)^2 = ∑ r : Fin n, (l r)^2 := by
  have hpair : ∀ i j : Fin n,
      ((if won i j then (1 : ℤ) else 0) + (if won j i then 1 else 0)) = if i = j then 0 else 1 := by
    intro i j
    by_cases hij : i = j
    · subst hij; simp [hirrefl]
    · have := hantisymm i j hij
      rw [if_neg hij]
      cases h : won j i <;> simp_all
  have hsum : 2 * ∑ r, w r = (n : ℤ) * (n - 1) := by
    have : ∑ r, w r = ∑ r, ∑ j, (if won j r then (1 : ℤ) else 0) := by
      rw [hw]; exact Finset.sum_comm
    calc 2 * ∑ r, w r = ∑ r, w r + ∑ r, ∑ j, (if won j r then (1 : ℤ) else 0) := by
          rw [← this]; ring
      _ = ∑ r, ∑ j, ((if won r j then (1 : ℤ) else 0) + (if won j r then 1 else 0)) := by
          rw [hw, ← Finset.sum_add_distrib]
          exact Finset.sum_congr rfl fun r _ => (Finset.sum_add_distrib).symm
      _ = ∑ r : Fin n, ∑ j : Fin n, (if r = j then (0 : ℤ) else 1) :=
          Finset.sum_congr rfl fun r _ => Finset.sum_congr rfl fun j _ => hpair r j
      _ = (n : ℤ) * (n - 1) := by
          have hrow : ∀ r : Fin n, ∑ j : Fin n, (if r = j then (0 : ℤ) else 1) = n - 1 := by
            intro r
            have e : ∀ j : Fin n, (if r = j then (0 : ℤ) else 1) = 1 - (if r = j then 1 else 0) :=
              fun j => by split_ifs <;> simp
            rw [Finset.sum_congr rfl fun j _ => e j, Finset.sum_sub_distrib, Finset.sum_ite_eq,
              if_pos (Finset.mem_univ r)]
            simp
          rw [Finset.sum_congr rfl fun r _ => hrow r, Finset.sum_const, Finset.card_univ,
            Fintype.card_fin, nsmul_eq_mul]
  have hexp : ∑ r, (l r) ^ 2 = ∑ r, (w r) ^ 2 + ((n : ℤ) * (n - 1) ^ 2 - (n - 1) * (2 * ∑ r, w r)) := by
    rw [hl]
    simp only
    rw [Finset.mul_sum, Finset.mul_sum]
    have : ∀ r, ((n : ℤ) - 1 - w r) ^ 2 = (w r) ^ 2 + ((n - 1) ^ 2 - (n - 1) * (2 * w r)) := fun r => by ring
    rw [Finset.sum_congr rfl fun r _ => this r, Finset.sum_add_distrib, Finset.sum_sub_distrib,
      Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]
  rw [hexp, hsum]
  ring
