theorem putnam_2010_a4
: ∀ n : ℕ, n > 0 → ¬Nat.Prime (10^10^10^n + 10^10^n + 10^n - 1) := by
  intro n hn hP
  obtain ⟨m, k, hnk, hk_odd⟩ : ∃ m k, n = 2 ^ m * k ∧ ¬ 2 ∣ k :=
    ⟨n.factorization 2, ordCompl[2] n,
     (Nat.ordProj_mul_ordCompl_eq_self n 2).symm,
     Nat.not_dvd_ordCompl Nat.prime_two (by omega)⟩
  set D : ℕ := 10 ^ 2 ^ m + 1 with hD
  have h2m_dvd : 2 ^ m ∣ n := ⟨k, hnk⟩
  have h2m_le : 2 ^ m ≤ n := Nat.le_of_dvd (by omega) h2m_dvd
  have hm_lt : m < 2 ^ m := Nat.lt_two_pow_self
  -- base congruence mod D
  have hDcast : (D : ℤ) = 10 ^ 2 ^ m + 1 := by rw [hD]; push_cast; ring
  have hbase : (10 : ℤ) ^ 2 ^ m ≡ -1 [ZMOD (D : ℤ)] := by
    rw [Int.modEq_iff_dvd, hDcast]
    exact ⟨-1, by ring⟩
  have hpow : ∀ q : ℕ, (10 : ℤ) ^ (2 ^ m * q) ≡ (-1) ^ q [ZMOD (D : ℤ)] := by
    intro q
    have h := hbase.pow q
    rwa [← pow_mul] at h
  -- 10^n ≡ -1
  have hCn : (10 : ℤ) ^ n ≡ -1 [ZMOD (D : ℤ)] := by
    have h := hpow k
    rw [← hnk] at h
    rwa [Odd.neg_one_pow (Nat.odd_iff.mpr (Nat.two_dvd_ne_zero.mp hk_odd))] at h
  -- exponents divisible by 2^(m+1) give ≡ 1
  have hpow1 : ∀ e : ℕ, 2 ^ (m + 1) ∣ e → (10 : ℤ) ^ e ≡ 1 [ZMOD (D : ℤ)] := by
    rintro e ⟨j, rfl⟩
    have h := hpow (2 * j)
    have he : 2 ^ m * (2 * j) = 2 ^ (m + 1) * j := by ring
    rw [he] at h
    simpa [pow_mul] using h
  have hdvd_10n : 2 ^ (m + 1) ∣ 10 ^ n :=
    (pow_dvd_pow 2 (by omega)).trans (pow_dvd_pow_of_dvd (by norm_num) n)
  have h10n_gt : n < 10 ^ n := Nat.lt_pow_self (by norm_num)
  have hdvd_1010n : 2 ^ (m + 1) ∣ 10 ^ 10 ^ n :=
    (pow_dvd_pow 2 (by omega)).trans (pow_dvd_pow_of_dvd (by norm_num) _)
  -- assemble: D divides the whole thing over ℤ
  have hA := hpow1 (10 ^ 10 ^ n) hdvd_1010n
  have hB := hpow1 (10 ^ n) hdvd_10n
  have hsum := (hA.add hB).add hCn
  have hzero := hsum.sub_right 1
  norm_num at hzero
  have hND : (D : ℤ) ∣ ((10:ℤ) ^ 10 ^ 10 ^ n + 10 ^ 10 ^ n + 10 ^ n - 1) :=
    Int.modEq_zero_iff_dvd.mp hzero
  -- back to ℕ
  have hC1 : 1 ≤ 10 ^ n := Nat.one_le_pow _ _ (by norm_num)
  have hND' : D ∣ 10 ^ 10 ^ 10 ^ n + 10 ^ 10 ^ n + 10 ^ n - 1 := by
    rw [← Int.natCast_dvd_natCast]
    have h1 : 1 ≤ 10 ^ 10 ^ 10 ^ n + 10 ^ 10 ^ n + 10 ^ n :=
      le_trans hC1 (Nat.le_add_left _ _)
    rw [Nat.cast_sub h1]
    push_cast
    exact hND
  -- size bounds: 1 < D < N
  clear hND hzero hsum hA hB hCn hbase hpow hpow1 hDcast
  have hX1 : 1 ≤ 10 ^ 2 ^ m := Nat.one_le_pow _ _ (by norm_num)
  have hXC : (10:ℕ) ^ 2 ^ m ≤ 10 ^ n := Nat.pow_le_pow_right (by norm_num) h2m_le
  have hCB : (10:ℕ) ^ n < 10 ^ 10 ^ n := Nat.lt_pow_self (by norm_num)
  have hBA : (10:ℕ) ^ 10 ^ n < 10 ^ 10 ^ 10 ^ n := Nat.lt_pow_self (by norm_num)
  rcases hP.eq_one_or_self_of_dvd D hND' with h | h
  · omega
  · omega
