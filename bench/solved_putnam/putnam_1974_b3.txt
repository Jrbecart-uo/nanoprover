import Mathlib

/-!
Putnam 1974 B3.  If `cos (π α) = 1/3` then `α` is irrational.

Elementary proof (no algebraic integers needed):
* `x k := cos (π α 2^k)` satisfies `x (k+1) = 2 (x k)^2 - 1` (double angle), `x 0 = 1/3`,
  so `x k = c k / 3^(2^k)` with `c 0 = 1`, `c (k+1) = 2 (c k)^2 - 3^(2^(k+1))`, and
  `3 ∤ c k` (mod 3: `c (k+1) ≡ 2 (c k)^2 ≢ 0`).
* Hence the `x k` are pairwise distinct: `x j = x k`, `j < k` would give
  `c k = 3^(2^k - 2^j) c j`, divisible by 3.
* If `α = a/b` were rational, `x k = cos (π (a 2^k mod 2b) / b)` takes at most `2b` values.
-/

/-- Numerators: `x k = c k / 3 ^ 2 ^ k`. -/
def c1974 : ℕ → ℤ
  | 0 => 1
  | k + 1 => 2 * c1974 k ^ 2 - 3 ^ 2 ^ (k + 1)

lemma c1974_not_dvd (k : ℕ) : ¬ (3 : ℤ) ∣ c1974 k := by
  rw [show (3 : ℤ) = ((3 : ℕ) : ℤ) by rfl, ← ZMod.intCast_zmod_eq_zero_iff_dvd]
  induction k with
  | zero => decide
  | succ k ih =>
    have h3 : ((3 : ℤ) : ZMod 3) ^ 2 ^ (k + 1) = 0 := by
      rw [show ((3 : ℤ) : ZMod 3) = 0 by decide]
      exact zero_pow (by positivity)
    have key : ∀ y : ZMod 3, y ≠ 0 → 2 * y ^ 2 ≠ 0 := by decide
    simp only [c1974, Int.cast_sub, Int.cast_mul, Int.cast_pow, Int.cast_ofNat] at h3 ⊢
    rw [show ((3 : ZMod 3)) ^ 2 ^ (k + 1) = 0 by simpa using h3, sub_zero]
    exact key _ ih

theorem putnam_1974_b3
(α : ℝ)
(ha : Real.cos (Real.pi * α) = (1 : ℝ)/3)
: Irrational α := by
  set x : ℕ → ℝ := fun k => Real.cos (Real.pi * α * 2 ^ k) with hx
  -- closed form
  have hform : ∀ k, x k = (c1974 k : ℝ) / 3 ^ 2 ^ k := by
    intro k
    induction k with
    | zero => simp [hx, c1974, ha]
    | succ k ih =>
      have : x (k + 1) = 2 * x k ^ 2 - 1 := by
        simp only [hx]
        rw [show Real.pi * α * 2 ^ (k + 1) = 2 * (Real.pi * α * 2 ^ k) by ring, Real.cos_two_mul]
      rw [this, ih]
      simp only [c1974, Int.cast_sub, Int.cast_mul, Int.cast_pow, Int.cast_ofNat]
      have h0 : (3 : ℝ) ^ 2 ^ k ≠ 0 := by positivity
      rw [pow_succ 2 k, pow_mul]
      field_simp
  -- injectivity
  have hinj : Function.Injective x := by
    have one_way : ∀ j k, j < k → x j ≠ x k := by
      intro j k hjk hjkx
      rw [hform, hform] at hjkx
      have hle : 2 ^ j ≤ 2 ^ k := Nat.pow_le_pow_right (by norm_num) hjk.le
      have hlt : 2 ^ j < 2 ^ k := Nat.pow_lt_pow_right (by norm_num) hjk
      have hsplit : (3 : ℝ) ^ 2 ^ k = 3 ^ 2 ^ j * 3 ^ (2 ^ k - 2 ^ j) := by
        rw [← pow_add, Nat.add_sub_cancel' hle]
      have h0 : (3 : ℝ) ^ 2 ^ j ≠ 0 := by positivity
      have hreal : (c1974 k : ℝ) = 3 ^ (2 ^ k - 2 ^ j) * c1974 j := by
        rw [div_eq_div_iff h0 (by positivity), hsplit] at hjkx
        have := mul_left_cancel₀ h0 (by linarith [hjkx] : (3 : ℝ) ^ 2 ^ j * (c1974 k) =
          3 ^ 2 ^ j * (3 ^ (2 ^ k - 2 ^ j) * c1974 j))
        exact this
      have hint : c1974 k = 3 ^ (2 ^ k - 2 ^ j) * c1974 j := by exact_mod_cast hreal
      apply c1974_not_dvd k
      rw [hint]
      exact Dvd.dvd.mul_right (dvd_pow_self 3 (by omega)) _
    intro j k h
    rcases lt_trichotomy j k with hjk | rfl | hkj
    · exact absurd h (one_way j k hjk)
    · rfl
    · exact absurd h.symm (one_way k j hkj)
  -- rationality would make the range finite
  rintro ⟨q, rfl⟩
  have hb : (0 : ℤ) < 2 * q.den := by have := q.den_pos; omega
  have hfin : (Set.range x).Finite := by
    apply (Set.finite_range (fun m : Fin (2 * q.den) => Real.cos (Real.pi * m / q.den))).subset
    rintro _ ⟨k, rfl⟩
    set M : ℤ := q.num * 2 ^ k
    have hr0 : 0 ≤ M % (2 * q.den) := Int.emod_nonneg _ hb.ne'
    have hr1 : M % (2 * q.den) < 2 * q.den := Int.emod_lt_of_pos _ hb
    refine ⟨⟨(M % (2 * q.den)).toNat, by omega⟩, ?_⟩
    have hden : (q.den : ℝ) ≠ 0 := by exact_mod_cast q.den_pos.ne'
    have hsplit := Int.emod_add_mul_ediv M (2 * q.den)
    simp only [hx]
    have harg : Real.pi * (q : ℝ) * 2 ^ k =
        Real.pi * ((M % (2 * q.den) : ℤ) : ℝ) / q.den + ((M / (2 * q.den) : ℤ) : ℝ) * (2 * Real.pi) := by
      have hq : (q : ℝ) = q.num / q.den := by exact_mod_cast (Rat.num_div_den q).symm
      have hM : ((M : ℤ) : ℝ) = ((M % (2 * q.den) : ℤ) : ℝ) + 2 * q.den * ((M / (2 * q.den) : ℤ) : ℝ) := by
        exact_mod_cast hsplit.symm
      rw [hq]
      field_simp
      have : ((M : ℤ) : ℝ) = q.num * 2 ^ k := by push_cast [M]; ring
      rw [← this, hM]
      ring
    rw [harg, Real.cos_add_int_mul_two_pi]
    congr 2
    have := Int.toNat_of_nonneg hr0
    congr 1
    exact_mod_cast this
  exact Set.infinite_range_of_injective hinj hfin
