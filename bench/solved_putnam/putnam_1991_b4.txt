theorem putnam_1991_b4
(p : ℕ)
(podd : Odd p)
(pprime : Prime p)
: (∑ j : Fin (p + 1), (p.choose j) * ((p + j).choose j)) ≡ (2 ^ p + 1) [MOD (p ^ 2)] := by
  have hp : Nat.Prime p := Nat.prime_iff.mpr pprime
  haveI := Fact.mk hp
  have hp2 : 2 ≤ p := hp.two_le
  have hp0 : p ≠ 0 := by omega
  -- p^2 = 0 in ZMod (p^2)
  have hpz : (p : ZMod (p ^ 2)) ^ 2 = 0 := by
    have h := ZMod.natCast_self (p ^ 2)
    push_cast at h
    exact h
  -- Lucas step: C(p+j, j) ≡ 1 mod p for j < p
  have hlucas : ∀ j, j < p → (p + j).choose j ≡ 1 [MOD p] := by
    intro j hj
    have h := Choose.choose_modEq_choose_mod_mul_choose_div_nat (n := p + j) (k := j) (p := p)
    have h1 : (p + j) % p = j := by rw [Nat.add_mod_left]; exact Nat.mod_eq_of_lt hj
    have h2 : j % p = j := Nat.mod_eq_of_lt hj
    have h3 : (p + j) / p = 1 := by
      rw [Nat.add_comm, Nat.add_div_right _ (by omega : 0 < p), Nat.div_eq_of_lt hj]
    have h4 : j / p = 0 := Nat.div_eq_of_lt hj
    rw [h1, h2, h3, h4, Nat.choose_self, Nat.choose_zero_right] at h
    simpa using h
  -- interior terms collapse: C(p,j)*C(p+j,j) = C(p,j) in ZMod (p^2)
  have hterm : ∀ j ∈ Finset.range p,
      (↑(p.choose j) * ↑((p + j).choose j) : ZMod (p ^ 2)) = ↑(p.choose j) := by
    intro j hj
    have hjp : j < p := Finset.mem_range.mp hj
    rcases Nat.eq_zero_or_pos j with rfl | hj0
    · simp
    · obtain ⟨u, hu⟩ := hp.dvd_choose_self (by omega) hjp
      have hC1 : 0 < (p + j).choose j := Nat.choose_pos (by omega)
      obtain ⟨w, hw⟩ := (Nat.modEq_iff_dvd' (by omega)).mp (hlucas j hjp).symm
      have hv : (p + j).choose j = 1 + p * w := by omega
      rw [hu, hv]
      push_cast
      linear_combination ((u : ZMod (p ^ 2)) * w) * hpz
  -- Babbage via Vandermonde: C(2p, p) = 2 in ZMod (p^2)
  have hmid : ∀ k ∈ Finset.range (p + 1),
      (↑(p.choose k) * ↑(p.choose (p - k)) : ZMod (p ^ 2))
      = (if k = 0 then 1 else 0) + (if k = p then 1 else 0) := by
    intro k hk
    have hk_le : k ≤ p := by have := Finset.mem_range.mp hk; omega
    rcases eq_or_ne k 0 with rfl | hk0
    · simp [Nat.choose_self, Ne.symm hp0]
    · rcases eq_or_ne k p with rfl | hkp
      · simp [Nat.choose_self, hk0]
      · obtain ⟨u, hu⟩ := hp.dvd_choose_self hk0 (by omega)
        obtain ⟨w, hw⟩ := hp.dvd_choose_self (by omega : p - k ≠ 0) (by omega)
        rw [hu, hw]
        push_cast
        simp only [if_neg hk0, if_neg hkp, add_zero]
        linear_combination ((u : ZMod (p ^ 2)) * w) * hpz
  have hB : (↑((p + p).choose p) : ZMod (p ^ 2)) = 2 := by
    rw [Nat.add_choose_eq p p p,
        Finset.Nat.sum_antidiagonal_eq_sum_range_succ_mk (fun ij => p.choose ij.1 * p.choose ij.2) p]
    push_cast
    rw [Finset.sum_congr rfl hmid, Finset.sum_add_distrib,
        Finset.sum_ite_eq' (Finset.range (p + 1)) 0 (fun _ => (1 : ZMod (p ^ 2))),
        Finset.sum_ite_eq' (Finset.range (p + 1)) p (fun _ => (1 : ZMod (p ^ 2)))]
    norm_num [Finset.mem_range]
  -- row sum: sum over range p of C(p,j), plus 1, is 2^p
  have hsum : (∑ j ∈ Finset.range p, (↑(p.choose j) : ZMod (p ^ 2))) + 1 = 2 ^ p := by
    have h := Nat.sum_range_choose p
    rw [Finset.sum_range_succ, Nat.choose_self] at h
    have h2 := congrArg (fun m : ℕ => (m : ZMod (p ^ 2))) h
    push_cast at h2
    exact h2
  -- assemble
  rw [Fin.sum_univ_eq_sum_range (fun j => p.choose j * ((p + j).choose j)) (p + 1)]
  rw [← ZMod.natCast_eq_natCast_iff]
  push_cast
  rw [Finset.sum_range_succ, Finset.sum_congr rfl hterm, Nat.choose_self]
  push_cast
  rw [one_mul, hB]
  linear_combination hsum
