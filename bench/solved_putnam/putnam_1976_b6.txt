/-!
Putnam 1976 B6 (quasiperfect numbers are odd squares).  Write N = 2^a b with b odd.  σ is multiplicative and
σ(2^a) = 2^{a+1} − 1 is odd, so σ(N) = 2N + 1 odd forces σ(b) odd; for odd non-square b the involution
d ↦ b/d on divisors has no fixed point and pairs odd divisors, so σ(b) would be even: b = c².  If a ≥ 1,
M = 2^{a+1} − 1 ≡ 3 (mod 4) divides σ(N) = (M + 1)c² + 1, hence M ∣ c² + 1; M has a prime factor
p ≡ 3 (mod 4) (a product of primes ≡ 1 is ≡ 1), and c² ≡ −1 (mod p) is impossible.  So a = 0, N = c².
-/

namespace P1976B6

def s (n : ℕ) : ℕ := ∑ d ∈ n.divisors, d

lemma s_mul {m n : ℕ} (h : m.Coprime n) : s (m * n) = s m * s n := by
  have := (ArithmeticFunction.isMultiplicative_sigma (k := 1)).map_mul_of_coprime h
  simpa [s, ArithmeticFunction.sigma_one_apply] using this

lemma s_two_pow (a : ℕ) : s (2 ^ a) + 1 = 2 ^ (a + 1) := by
  rw [s, Nat.sum_divisors_prime_pow Nat.prime_two]
  induction a with
  | zero => simp
  | succ a ih => rw [sum_range_succ, pow_succ 2 (a + 1)]; omega

lemma even_s_of_not_sq {b : ℕ} (hb : Odd b) (hsq : ¬ IsSquare b) : Even (s b) := by
  rw [← ZMod.natCast_eq_zero_iff_even, s, Nat.cast_sum]
  have hb0 : b ≠ 0 := by rintro rfl; exact hsq ⟨0, rfl⟩
  refine Finset.sum_involution (fun d _ => b / d) ?_ ?_ ?_ ?_
  · intro d hd
    have hdb : d ∣ b := Nat.dvd_of_mem_divisors hd
    have h1 : Odd d := hb.of_dvd_nat hdb
    have h2 : Odd (b / d) := hb.of_dvd_nat (Nat.div_dvd_of_dvd hdb)
    rw [ZMod.natCast_eq_one_iff_odd.mpr h1, ZMod.natCast_eq_one_iff_odd.mpr h2]
    decide
  · intro d hd _ heq
    have hdb : d ∣ b := Nat.dvd_of_mem_divisors hd
    apply hsq
    refine ⟨d, ?_⟩
    have := Nat.div_mul_cancel hdb
    rw [heq] at this
    exact this.symm
  · intro d hd
    exact Nat.mem_divisors.mpr ⟨Nat.div_dvd_of_dvd (Nat.dvd_of_mem_divisors hd), hb0⟩
  · intro d hd
    exact Nat.div_div_self (Nat.dvd_of_mem_divisors hd) hb0

lemma mod4 (n : ℕ) : (∀ p, p.Prime → p ∣ n → p % 4 = 1) → n % 4 = 1 := by
  refine Nat.recOnMul (motive := fun n => (∀ p, p.Prime → p ∣ n → p % 4 = 1) → n % 4 = 1) ?_ ?_ ?_ ?_ n
  · intro h; have := h 2 Nat.prime_two (dvd_zero 2); omega
  · intro _; rfl
  · intro p hp h; exact h p hp dvd_rfl
  · intro a b iha ihb h
    have ha := iha (fun p hp hd => h p hp (dvd_mul_of_dvd_left hd b))
    have hb := ihb (fun p hp hd => h p hp (dvd_mul_of_dvd_right hd a))
    rw [Nat.mul_mod, ha, hb]

end P1976B6

open P1976B6 in
theorem putnam_1976_b6
(σ : ℕ → ℤ)
(hσ : σ = fun N : ℕ => ∑ d ∈ Nat.divisors N, (d : ℤ))
(quasiperfect : ℕ → Prop)
(quasiperfect_def : ∀ N, quasiperfect N ↔ σ N = 2*N + 1)
: ∀ N : ℕ, quasiperfect N → ∃ m : ℤ, Odd m ∧ m^2 = N := by
  intro N hN
  rw [quasiperfect_def, hσ] at hN
  have hsN : s N = 2 * N + 1 := by
    have : ((s N : ℕ) : ℤ) = ((2 * N + 1 : ℕ) : ℤ) := by simp only [s, Nat.cast_sum]; push_cast; exact hN
    exact_mod_cast this
  have hN0 : N ≠ 0 := by rintro rfl; simp [s] at hsN
  obtain ⟨a, b, hbodd, rfl⟩ := Nat.exists_eq_two_pow_mul_odd hN0
  have hcop : Nat.Coprime (2 ^ a) b := (hbodd.coprime_two_right.symm).pow_left a
  have hmul := s_mul hcop
  have h2a := s_two_pow a
  have hsb_odd : Odd (s b) := by
    by_contra h
    rw [Nat.not_odd_iff_even] at h
    have h1 : Even (s (2 ^ a * b)) := by rw [hmul]; exact h.mul_left _
    have h2 : Odd (s (2 ^ a * b)) := ⟨2 ^ a * b, by rw [hsN]⟩
    exact (Nat.not_even_iff_odd.mpr h2) h1
  have hsq : IsSquare b := by
    by_contra h
    exact (Nat.not_even_iff_odd.mpr hsb_odd) (even_s_of_not_sq hbodd h)
  obtain ⟨c, rfl⟩ := hsq
  have hcodd : Odd c := (Nat.odd_mul.mp hbodd).1
  rcases Nat.eq_zero_or_pos a with rfl | ha
  · refine ⟨c, by exact_mod_cast hcodd, by push_cast; ring⟩
  · exfalso
    set M := s (2 ^ a) with hMdef
    have hdiv : M ∣ c * c + 1 := by
      have h1 : M ∣ 2 * (2 ^ a * (c * c)) + 1 := by rw [← hsN, hmul]; exact dvd_mul_right _ _
      have h22 : 2 * 2 ^ a = M + 1 := by rw [h2a, pow_succ]; ring
      have e : 2 * (2 ^ a * (c * c)) + 1 = M * (c * c) + (c * c + 1) := by
        calc 2 * (2 ^ a * (c * c)) + 1 = (2 * 2 ^ a) * (c * c) + 1 := by ring
          _ = (M + 1) * (c * c) + 1 := by rw [h22]
          _ = _ := by ring
      rw [e] at h1
      exact (Nat.dvd_add_right (dvd_mul_right _ _)).mp h1
    have hM4 : M % 4 = 3 := by
      have h4 : 4 ∣ 2 ^ (a + 1) := by
        rw [show 4 = 2 ^ 2 by norm_num]; exact Nat.pow_dvd_pow 2 (by omega)
      omega
    obtain ⟨p, hp, hpM, hp3⟩ : ∃ p, p.Prime ∧ p ∣ M ∧ p % 4 = 3 := by
      by_contra hne
      push Not at hne
      have := mod4 M (fun p hp hd => by
        have h3 := hne p hp hd
        have hpodd : p % 2 = 1 := by
          rcases hp.eq_two_or_odd with h | h
          · subst h; omega
          · exact h
        omega)
      omega
    haveI := Fact.mk hp
    have hsq' : ((c : ZMod p)) ^ 2 = -1 := by
      have hd : p ∣ c * c + 1 := dvd_trans hpM hdiv
      have h0 : ((c * c + 1 : ℕ) : ZMod p) = 0 := (ZMod.natCast_eq_zero_iff _ _).mpr hd
      push_cast at h0
      linear_combination h0
    exact ZMod.mod_four_ne_three_of_sq_eq_neg_one hsq' hp3
