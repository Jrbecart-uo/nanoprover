/-!
Putnam 1967 B2.  Evaluating the identities at `(1,0), (0,1), (1,1)`:
`A = p², C = (1-p)², B = 2p(1-p)` and `α = pr, γ = (1-p)(1-r), β = p + r - 2pr`.
If `A, C < 4/9` then `p ∈ (1/3, 2/3)` so `B > 4/9`.  With `u = p - 1/2, v = r - 1/2`:
`β < 4/9 ⇒ uv > 1/36`, while `α, γ < 4/9 ⇒ |u+v|/2 < 7/36 - uv`, and `(u+v)² ≥ 4uv`
then forces `uv < 1/36` — contradiction.
-/
theorem putnam_1967_b2
(p r A B C α β γ : ℝ)
(prbound : 0 ≤ p ∧ p ≤ 1 ∧ 0 ≤ r ∧ r ≤ 1)
(id1 : ∀ x y : ℝ, (p * x + (1 - p) * y) ^ 2 = A * x ^ 2 + B * x * y + C * y ^ 2)
(id2 : ∀ x y : ℝ, (p * x + (1 - p) * y) * (r * x + (1 - r) * y) = α * x ^ 2 + β * x * y + γ * y ^ 2)
: max (max A B) C ≥ 4 / 9 ∧ max (max α β) γ ≥ 4 / 9 := by
  have hA := id1 1 0
  have hC := id1 0 1
  have hB := id1 1 1
  have hα := id2 1 0
  have hγ := id2 0 1
  have hβ := id2 1 1
  norm_num at hA hC hB hα hγ hβ
  constructor
  · by_contra h
    simp only [ge_iff_le, le_max_iff, not_or, not_le] at h
    obtain ⟨⟨h1, h2⟩, h3⟩ := h
    nlinarith [sq_nonneg (p - 1/2), sq_nonneg (p - 1/3), sq_nonneg (p - 2/3)]
  · by_contra h
    simp only [ge_iff_le, le_max_iff, not_or, not_le] at h
    obtain ⟨⟨h1, h2⟩, h3⟩ := h
    -- m = uv with u = p - 1/2, v = r - 1/2
    have hm1 : (p - 1/2) * (r - 1/2) > 1/36 := by nlinarith
    have hw1 : (p - 1/2 + (r - 1/2)) / 2 < 7/36 - (p - 1/2) * (r - 1/2) := by nlinarith
    have hw2 : -(p - 1/2 + (r - 1/2)) / 2 < 7/36 - (p - 1/2) * (r - 1/2) := by nlinarith
    have hsq : (p - 1/2 + (r - 1/2)) ^ 2 ≥ 4 * ((p - 1/2) * (r - 1/2)) := by
      nlinarith [sq_nonneg (p - r)]
    nlinarith [sq_nonneg (p - 1/2 + (r - 1/2)), mul_pos (by linarith : (0:ℝ) < 7/36 - (p - 1/2) * (r - 1/2) - (p - 1/2 + (r - 1/2)) / 2) (by linarith : (0:ℝ) < 7/36 - (p - 1/2) * (r - 1/2) + (p - 1/2 + (r - 1/2)) / 2)]
