import Mathlib

open Set Function

/-!
Putnam 1964 A4.  A bounded integer sequence with `uₙ` determined by `uₙ₋₁, …, uₙ₋₄` is
eventually periodic: the state `(uₙ, uₙ₊₁, uₙ₊₂, uₙ₊₃)` lives in the finite box `[B,T]⁴`, so
two states coincide (pigeonhole), and the recurrence propagates the coincidence forever.
-/
theorem putnam_1964_a4
(u : ℕ → ℤ)
(boundedu : ∃ B T : ℤ, ∀ n : ℕ, B ≤ u n ∧ u n ≤ T)
(hu : ∀ n ≥ 4, u n = ((u (n - 1) + u (n - 2) + u (n - 3) * u (n - 4)) : ℝ) / (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ∧ (u (n - 1) * u (n - 2) + u (n - 3) + u (n - 4)) ≠ 0)
: (∃ N c : ℕ, c > 0 ∧ ∀ n ≥ N, u (n + c) = u n) := by
  obtain ⟨B, T, hB⟩ := boundedu
  let st : ℕ → ℤ × ℤ × ℤ × ℤ := fun n => (u n, u (n + 1), u (n + 2), u (n + 3))
  let S : Set (ℤ × ℤ × ℤ × ℤ) := Icc B T ×ˢ Icc B T ×ˢ Icc B T ×ˢ Icc B T
  have hmaps : MapsTo st univ S := fun n _ => by
    simp only [st, S, Set.mem_prod, Set.mem_Icc]
    exact ⟨hB n, hB (n + 1), hB (n + 2), hB (n + 3)⟩
  have hfin : S.Finite :=
    (finite_Icc B T).prod ((finite_Icc B T).prod ((finite_Icc B T).prod (finite_Icc B T)))
  -- the recurrence, re-indexed
  have hnext : ∀ m, (u (m + 4) : ℝ) = ((u (m + 3) + u (m + 2) + u (m + 1) * u m) : ℝ) /
      (u (m + 3) * u (m + 2) + u (m + 1) + u m) := by
    intro m
    have := (hu (m + 4) (by omega)).1
    rw [show m + 4 - 1 = m + 3 by omega, show m + 4 - 2 = m + 2 by omega,
      show m + 4 - 3 = m + 1 by omega, show m + 4 - 4 = m by omega] at this
    exact this
  have periodic : ∀ i j, i < j → st i = st j → ∃ N c : ℕ, c > 0 ∧ ∀ n ≥ N, u (n + c) = u n := by
    intro i j hij heq
    have hk : ∀ k t, t < 4 → u (i + k + t) = u (j + k + t) := by
      intro k
      induction k with
      | zero =>
        intro t ht
        simp only [st, Prod.mk.injEq] at heq
        obtain ⟨e0, e1, e2, e3⟩ := heq
        interval_cases t <;> simpa
      | succ k ih =>
        intro t ht
        by_cases h3 : t < 3
        · have := ih (t + 1) (by omega)
          rwa [show i + (k + 1) + t = i + k + (t + 1) by omega,
            show j + (k + 1) + t = j + k + (t + 1) by omega]
        · obtain rfl : t = 3 := by omega
          rw [show i + (k + 1) + 3 = i + k + 4 by omega, show j + (k + 1) + 3 = j + k + 4 by omega]
          have e0 := ih 0 (by norm_num)
          have e1 := ih 1 (by norm_num)
          have e2 := ih 2 (by norm_num)
          have e3 := ih 3 (by norm_num)
          simp only [add_zero] at e0
          have : (u (i + k + 4) : ℝ) = u (j + k + 4) := by
            rw [hnext (i + k), hnext (j + k), e0, e1, e2, e3]
          exact_mod_cast this
    refine ⟨i, j - i, by omega, fun n hn => ?_⟩
    obtain ⟨k, rfl⟩ : ∃ k, n = i + k := ⟨n - i, by omega⟩
    have := hk k 0 (by norm_num)
    simp only [add_zero] at this
    rw [show i + k + (j - i) = j + k by omega]
    exact this.symm
  obtain ⟨i, -, j, -, hij, heq⟩ := Set.infinite_univ.exists_ne_map_eq_of_mapsTo hmaps hfin
  rcases lt_or_gt_of_ne hij with h | h
  · exact periodic i j h heq
  · exact periodic j i h heq.symm
