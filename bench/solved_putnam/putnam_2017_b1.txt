/-!
Putnam 2017 B1.  (→) Rebase both lines at a common point X.  Distinct lines through X have directions with
det(w₁,w₂) ≠ 0 (det = 0 makes w₁ a nonzero multiple of w₂, hence the same line), so (λ−1)(X−P) = α w₁ + β w₂
by Cramer, and A₁ = X − (α/λ) w₁, A₂ = X + β w₂ work.  (←) If the lines were disjoint, the midpoint P of the
base points v₁, v₂ lies on neither (P ∈ L₁ would put v₂ on L₁); λ = 1 then forces A₁ = A₂ ∈ L₁ ∩ L₂.
-/

namespace P2017B1

def det (a b : Fin 2 → ℝ) : ℝ := a 0 * b 1 - a 1 * b 0

lemma line_rebase {v w X : Fin 2 → ℝ} (hX : X ∈ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v + t • w}) :
    {p : Fin 2 → ℝ | ∃ t : ℝ, p = v + t • w} = {p | ∃ t : ℝ, p = X + t • w} := by
  obtain ⟨t0, rfl⟩ := hX
  ext p
  constructor
  · rintro ⟨t, rfl⟩
    exact ⟨t - t0, by ext i; simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul]; ring⟩
  · rintro ⟨t, rfl⟩
    exact ⟨t0 + t, by ext i; simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul]; ring⟩

lemma ne_zero_iff' {w : Fin 2 → ℝ} (hw : w ≠ 0) : w 0 ≠ 0 ∨ w 1 ≠ 0 := by
  by_contra h
  push Not at h
  exact hw (by ext i; fin_cases i <;> simp [h.1, h.2])

lemma line_smul {X w : Fin 2 → ℝ} {c : ℝ} (hc : c ≠ 0) :
    {p : Fin 2 → ℝ | ∃ t : ℝ, p = X + t • (c • w)} = {p | ∃ t : ℝ, p = X + t • w} := by
  ext p
  constructor
  · rintro ⟨t, rfl⟩; exact ⟨t * c, by rw [smul_smul]⟩
  · rintro ⟨t, rfl⟩; exact ⟨t / c, by rw [smul_smul, div_mul_cancel₀ _ hc]⟩

lemma eq_of_det_zero {X w1 w2 : Fin 2 → ℝ} (h1 : w1 ≠ 0) (h2 : w2 ≠ 0) (hd : det w1 w2 = 0) :
    {p : Fin 2 → ℝ | ∃ t : ℝ, p = X + t • w1} = {p | ∃ t : ℝ, p = X + t • w2} := by
  unfold det at hd
  have key : ∃ c : ℝ, c ≠ 0 ∧ w1 = c • w2 := by
    rcases ne_zero_iff' h2 with h | h
    · refine ⟨w1 0 / w2 0, ?_, ?_⟩
      · intro hc
        have e0 : w1 0 = 0 := by rw [div_eq_zero_iff] at hc; tauto
        have e1 : w1 1 = 0 := by
          have : w1 1 * w2 0 = 0 := by rw [e0] at hd; linarith
          rcases mul_eq_zero.mp this with h' | h'
          · exact h'
          · exact absurd h' h
        exact h1 (by ext i; fin_cases i <;> simp [e0, e1])
      · ext i; fin_cases i
        · simp; field_simp
        · simp; field_simp; linarith
    · have e0 : w2 0 = 0 ∨ w2 0 ≠ 0 := em _
      refine ⟨w1 1 / w2 1, ?_, ?_⟩
      · intro hc
        have e1 : w1 1 = 0 := by rw [div_eq_zero_iff] at hc; tauto
        have e0' : w1 0 = 0 := by
          have : w1 0 * w2 1 = 0 := by rw [e1] at hd; linarith
          rcases mul_eq_zero.mp this with h' | h'
          · exact h'
          · exact absurd h' h
        exact h1 (by ext i; fin_cases i <;> simp [e0', e1])
      · ext i; fin_cases i
        · simp; field_simp; linarith
        · simp; field_simp
  obtain ⟨c, hc, rfl⟩ := key
  exact line_smul hc

lemma solve {w1 w2 : Fin 2 → ℝ} (hD : det w1 w2 ≠ 0) (u : Fin 2 → ℝ) :
    ∃ α β : ℝ, u = α • w1 + β • w2 := by
  refine ⟨det u w2 / det w1 w2, det w1 u / det w1 w2, ?_⟩
  unfold det at hD ⊢
  have hD' : w2 1 * w1 0 - w2 0 * w1 1 ≠ 0 := by contrapose! hD; linarith
  ext i; fin_cases i
  · simp; field_simp; ring
  · simp; field_simp; ring

end P2017B1

open P2017B1 in
theorem putnam_2017_b1
  (lines : Set (Set (Fin 2 → ℝ)))
  (L1 L2 : Set (Fin 2 → ℝ))
  (L1L2lines : L1 ∈ lines ∧ L2 ∈ lines)
  (L1L2distinct : L1 ≠ L2)
  (hlines : lines = {L | ∃ v w : Fin 2 → ℝ, w ≠ 0 ∧ L = {p | ∃ t : ℝ, p = v + t • w}}) :
  L1 ∩ L2 ≠ ∅ ↔ (∀ lambda : ℝ, lambda ≠ 0 →
    ∀ P, (P ∉ L1 ∧ P ∉ L2) → ∃ A1 A2, A1 ∈ L1 ∧ A2 ∈ L2 ∧ (A2 - P = lambda • (A1 - P))) := by
  subst hlines
  obtain ⟨⟨v1, w1, hw1, rfl⟩, ⟨v2, w2, hw2, rfl⟩⟩ := L1L2lines
  constructor
  · intro hne lam hlam P hP
    obtain ⟨X, hX1, hX2⟩ := Set.nonempty_iff_ne_empty.mpr hne
    rw [line_rebase hX1] at L1L2distinct ⊢
    rw [line_rebase hX2] at L1L2distinct ⊢
    have hD : det w1 w2 ≠ 0 := fun hd => L1L2distinct (eq_of_det_zero hw1 hw2 hd)
    obtain ⟨α, β, hu⟩ := solve hD ((lam - 1) • (X - P))
    refine ⟨X + (-α / lam) • w1, X + β • w2, ⟨-α / lam, rfl⟩, ⟨β, rfl⟩, ?_⟩
    ext i
    have := congrFun hu i
    simp only [Pi.add_apply, Pi.sub_apply, Pi.smul_apply, smul_eq_mul] at this ⊢
    field_simp
    linarith
  · intro h hempty
    set P : Fin 2 → ℝ := (1 / 2 : ℝ) • (v1 + v2) with hPdef
    have hv1 : v1 ∈ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} := ⟨0, by simp⟩
    have hv2 : v2 ∈ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v2 + t • w2} := ⟨0, by simp⟩
    have hP1 : P ∉ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} := by
      rintro ⟨t, ht⟩
      have : v2 ∈ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} := by
        refine ⟨2 * t, ?_⟩
        ext i
        have := congrFun ht i
        simp only [hPdef, Pi.add_apply, Pi.smul_apply, smul_eq_mul] at this ⊢
        linarith
      have hmem : v2 ∈ ({p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} ∩ {p | ∃ t : ℝ, p = v2 + t • w2}) := ⟨this, hv2⟩
      rw [hempty] at hmem; exact hmem
    have hP2 : P ∉ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v2 + t • w2} := by
      rintro ⟨t, ht⟩
      have : v1 ∈ {p : Fin 2 → ℝ | ∃ t : ℝ, p = v2 + t • w2} := by
        refine ⟨2 * t, ?_⟩
        ext i
        have := congrFun ht i
        simp only [hPdef, Pi.add_apply, Pi.smul_apply, smul_eq_mul] at this ⊢
        linarith
      have hmem : v1 ∈ ({p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} ∩ {p | ∃ t : ℝ, p = v2 + t • w2}) := ⟨hv1, this⟩
      rw [hempty] at hmem; exact hmem
    obtain ⟨A1, A2, hA1, hA2, heq⟩ := h 1 one_ne_zero P ⟨hP1, hP2⟩
    rw [one_smul, sub_left_inj] at heq
    have hmem : A1 ∈ ({p : Fin 2 → ℝ | ∃ t : ℝ, p = v1 + t • w1} ∩ {p | ∃ t : ℝ, p = v2 + t • w2}) := ⟨hA1, heq ▸ hA2⟩
    rw [hempty] at hmem; exact hmem
