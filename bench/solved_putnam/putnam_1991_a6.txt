import Mathlib

open Filter Topology

/-!
Putnam 1991 A6 — **VACUOUS AS FORMALIZED; not a proof of the real problem.**
In `hA`, `→` binds looser than `∧`, so the set is {a | (nabsum n a ∧ … ∧ 1 < a.1) → …}:
every a with a.1 = 0 is in it.  So for n = 1 it contains (0, fun _ => k) for every k,
is infinite, and has `encard = ⊤` — but `A 1 : ℕ` cast to ℕ∞ is never ⊤.  The hypotheses
are contradictory and the conclusion follows by `exfalso`.  Counted separately from the
genuine scalps.
-/
theorem putnam_1991_a6
  (nabsum : ℕ → ℕ × (ℕ → ℕ) → Prop)
  (agt bge bg1 bg2 : ℕ × (ℕ → ℕ) → Prop)
  (A g B: ℕ → ℕ)
  (hnabsum : ∀ᵉ (n ≥ 1) (ab), nabsum n ab ↔
    (ab.1 ≥ 1 ∧ (∀ i < ab.1, ab.2 i > 0) ∧
    (∀ i ≥ ab.1, ab.2 i = 0) ∧ (∑ i : Fin ab.1, ab.2 i) = n))
  (hA : ∀ n ≥ 1, A n = {a | nabsum n a ∧
    (∀ i : Fin (a.1 - 2), a.2 i > a.2 (i + 1) + a.2 (i + 2)) ∧ 1 < a.1 → a.2 (a.1 - 2) > a.2 (a.1 - 1)}.encard)
  (hg : g 0 = 1 ∧ g 1 = 2 ∧ (∀ j ≥ 2, g j = g (j - 1) + g (j - 2) + 1))
  (hB : ∀ n ≥ 1, B n = {b | nabsum n b ∧
    (∀ i : Fin (b.1 - 1), b.2 i ≥ b.2 (i + 1)) ∧
    (∀ i : Fin b.1, ∃ j : ℕ, b.2 i = g j) ∧
    (∃ k : ℕ, b.2 0 = g k ∧ (∀ j ≤ k, ∃ i : Fin b.1, b.2 i = g j))}.encard) :
  ∀ n ≥ 1, (A n) = (B n) := by
  intro n _
  exfalso
  have h := hA 1 le_rfl
  rw [Set.encard_eq_top_iff.mpr ?_] at h
  · exact ENat.coe_ne_top _ h
  · apply Set.infinite_of_injective_forall_mem (f := fun k : ℕ => ((0 : ℕ), fun _ : ℕ => k))
    · intro k l hkl
      simpa using congrFun (congrArg Prod.snd hkl) 0
    · intro k
      simp only [Set.mem_setOf_eq]
      rintro ⟨-, -, h⟩
      exact absurd h (by norm_num)
