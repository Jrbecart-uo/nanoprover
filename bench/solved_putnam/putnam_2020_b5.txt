private lemma putnam_2020_b5_key (u0 u1 u2 u3 : ℂ)
    (h0 : u0 ≠ 0) (h1 : u1 ≠ 0) (h2 : u2 ≠ 0) (h3 : u3 ≠ 0)
    (hr0 : u0.re = 1/2) (hr1 : u1.re = 1/2) (hr2 : u2.re = 1/2) (hr3 : u3.re = 1/2) :
    3 - (1 - u0⁻¹) - (1 - u1⁻¹) - (1 - u2⁻¹) - (1 - u3⁻¹)
      + (1 - u0⁻¹) * (1 - u1⁻¹) * (1 - u2⁻¹) * (1 - u3⁻¹) ≠ 0 := by
  intro hE
  have hEQ : (3 - (1 - u0⁻¹) - (1 - u1⁻¹) - (1 - u2⁻¹) - (1 - u3⁻¹)
      + (1 - u0⁻¹) * (1 - u1⁻¹) * (1 - u2⁻¹) * (1 - u3⁻¹)) * (u0 * u1 * u2 * u3)
      = (u0*u1 + u0*u2 + u0*u3 + u1*u2 + u1*u3 + u2*u3) - (u0 + u1 + u2 + u3) + 1 := by
    field_simp
    ring
  rw [hE, zero_mul] at hEQ
  have hQ : (u0*u1 + u0*u2 + u0*u3 + u1*u2 + u1*u3 + u2*u3)
      - (u0 + u1 + u2 + u3) + 1 = 0 := hEQ.symm
  have hu0f : u0 = (↑(1/2 : ℝ) : ℂ) + ↑u0.im * Complex.I := by
    have h := (Complex.re_add_im u0).symm
    rwa [hr0] at h
  have hu1f : u1 = (↑(1/2 : ℝ) : ℂ) + ↑u1.im * Complex.I := by
    have h := (Complex.re_add_im u1).symm
    rwa [hr1] at h
  have hu2f : u2 = (↑(1/2 : ℝ) : ℂ) + ↑u2.im * Complex.I := by
    have h := (Complex.re_add_im u2).symm
    rwa [hr2] at h
  have hu3f : u3 = (↑(1/2 : ℝ) : ℂ) + ↑u3.im * Complex.I := by
    have h := (Complex.re_add_im u3).symm
    rwa [hr3] at h
  rw [hu0f, hu1f, hu2f, hu3f] at hQ
  have hQre := congrArg Complex.re hQ
  have hQim := congrArg Complex.im hQ
  simp [Complex.mul_re, Complex.mul_im] at hQre hQim
  nlinarith [hQre, hQim, sq_nonneg (u0.im + u1.im + u2.im + u3.im),
    sq_nonneg u0.im, sq_nonneg u1.im, sq_nonneg u2.im, sq_nonneg u3.im]

theorem putnam_2020_b5
(z : Fin 4 → ℂ)
(hzle1 : ∀ n, ‖z n‖ = 1)
(hzne1 : ∀ n, z n ≠ 1)
: 3 - z 0 - z 1 - z 2 - z 3 + (z 0) * (z 1) * (z 2) * (z 3) ≠ 0 := by
  have h1z : ∀ n, (1 : ℂ) - z n ≠ 0 := fun n => sub_ne_zero.mpr (Ne.symm (hzne1 n))
  have hnormSq : ∀ n, Complex.normSq (z n) = 1 := by
    intro n
    rw [Complex.normSq_eq_norm_sq, hzle1 n, one_pow]
  have hxne : ∀ n, (z n).re ≠ 1 := by
    intro n hx
    apply hzne1 n
    have h := hnormSq n
    rw [Complex.normSq_apply, hx] at h
    have him0 : (z n).im * (z n).im = 0 := by linarith
    exact Complex.ext (by rw [hx, Complex.one_re])
      (by rw [mul_self_eq_zero.mp him0, Complex.one_im])
  have hre : ∀ n, (((1 : ℂ) - z n)⁻¹).re = 1/2 := by
    intro n
    rw [Complex.inv_re]
    have hsub_re : ((1 : ℂ) - z n).re = 1 - (z n).re := by simp
    have hsub_im : ((1 : ℂ) - z n).im = -(z n).im := by simp
    have h2 : Complex.normSq (1 - z n) = 2 * (1 - (z n).re) := by
      rw [Complex.normSq_apply, hsub_re, hsub_im]
      have h4 := hnormSq n
      rw [Complex.normSq_apply] at h4
      linear_combination h4
    have h5 : 1 - (z n).re ≠ 0 := sub_ne_zero.mpr (Ne.symm (hxne n))
    have h6 : (2 : ℝ) * (1 - (z n).re) ≠ 0 := mul_ne_zero two_ne_zero h5
    rw [hsub_re, h2, div_eq_iff h6]
    ring
  have hkey := putnam_2020_b5_key ((1 - z 0)⁻¹) ((1 - z 1)⁻¹) ((1 - z 2)⁻¹) ((1 - z 3)⁻¹)
    (inv_ne_zero (h1z 0)) (inv_ne_zero (h1z 1)) (inv_ne_zero (h1z 2)) (inv_ne_zero (h1z 3))
    (hre 0) (hre 1) (hre 2) (hre 3)
  have he : ∀ n, 1 - (((1 : ℂ) - z n)⁻¹)⁻¹ = z n := by
    intro n
    rw [inv_inv]
    ring
  rwa [he 0, he 1, he 2, he 3] at hkey
