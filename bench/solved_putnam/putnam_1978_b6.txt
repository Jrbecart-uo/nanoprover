import Mathlib

open Set Real Filter Topology Polynomial

/-!
Putnam 1978 B6.  Let xᵢ = (Σ_{j ≤ mi} aᵢⱼ)/i ∈ [0, m].  The claim is (Σ xᵢ)² ≤ 2m Σ i·xᵢ,
proved by induction on n: with T = Σ_{i≤n} xᵢ ≤ mn and 0 ≤ x ≤ m,
(T + x)² = T² + 2Tx + x² ≤ 2mW + 2mnx + mx ≤ 2m(W + (n+1)x)  (`nlinarith`).
-/
theorem putnam_1978_b6
(a : ℕ → ℕ → ℝ)
(ha : ∀ i j, a i j ∈ Icc 0 1)
(m n : ℕ)
(mnpos : m > 0 ∧ n > 0)
: ((∑ i ∈ Finset.Icc 1 n, ∑ j ∈ Finset.Icc 1 (m * i), a i j / i) ^ 2 ≤ 2 * m * ∑ i ∈ Finset.Icc 1 n, ∑ j ∈ Finset.Icc 1 (m * i), a i j) := by
  -- core: 0 ≤ xᵢ ≤ m ⇒ (Σ xᵢ)² ≤ 2m Σ i·xᵢ
  have core : ∀ (x : ℕ → ℝ), (∀ i, 1 ≤ i → 0 ≤ x i ∧ x i ≤ m) → ∀ N : ℕ,
      (∑ i ∈ Finset.Icc 1 N, x i) ^ 2 ≤ 2 * m * ∑ i ∈ Finset.Icc 1 N, (i : ℝ) * x i := by
    intro x hx N
    induction N with
    | zero => simp
    | succ N ih =>
      rw [Finset.sum_Icc_succ_top (by omega), Finset.sum_Icc_succ_top (by omega)]
      obtain ⟨h0, hm⟩ := hx (N + 1) (by omega)
      have hT0 : 0 ≤ ∑ i ∈ Finset.Icc 1 N, x i :=
        Finset.sum_nonneg fun i hi => (hx i (Finset.mem_Icc.mp hi).1).1
      have hT : ∑ i ∈ Finset.Icc 1 N, x i ≤ m * N := by
        have := Finset.sum_le_card_nsmul (Finset.Icc 1 N) x (m : ℝ)
          (fun i hi => (hx i (Finset.mem_Icc.mp hi).1).2)
        simpa [mul_comm] using this
      push_cast
      nlinarith [mul_le_mul_of_nonneg_right hT h0, mul_le_mul_of_nonneg_left hm h0]
  let x : ℕ → ℝ := fun i => (∑ j ∈ Finset.Icc 1 (m * i), a i j) / i
  have hx : ∀ i, 1 ≤ i → 0 ≤ x i ∧ x i ≤ m := by
    intro i hi
    have hi' : (0 : ℝ) < i := by exact_mod_cast hi
    have hS0 : 0 ≤ ∑ j ∈ Finset.Icc 1 (m * i), a i j := Finset.sum_nonneg fun j _ => (ha i j).1
    have hS : ∑ j ∈ Finset.Icc 1 (m * i), a i j ≤ m * i := by
      have := Finset.sum_le_card_nsmul (Finset.Icc 1 (m * i)) (a i) 1 (fun j _ => (ha i j).2)
      simpa using this
    refine ⟨div_nonneg hS0 hi'.le, ?_⟩
    rw [div_le_iff₀ hi']; exact hS
  have e1 : ∑ i ∈ Finset.Icc 1 n, ∑ j ∈ Finset.Icc 1 (m * i), a i j / i
      = ∑ i ∈ Finset.Icc 1 n, x i := by
    refine Finset.sum_congr rfl fun i _ => ?_
    simp only [x]; rw [Finset.sum_div]
  have e2 : ∑ i ∈ Finset.Icc 1 n, ∑ j ∈ Finset.Icc 1 (m * i), a i j
      = ∑ i ∈ Finset.Icc 1 n, (i : ℝ) * x i := by
    refine Finset.sum_congr rfl fun i hi => ?_
    have hi' : (i : ℝ) ≠ 0 := by
      have := (Finset.mem_Icc.mp hi).1; positivity
    simp only [x]; field_simp
  rw [e1, e2]
  exact core x hx n
