import Mathlib

/-!
Putnam 2018 B6.  Weight a tuple s by ∏ᵢ (1/2)^(sᵢ).  Every s ∈ S has weight (1/2)^3860, and
S ⊆ V^2018 for V = {1,2,3,4,5,6,10}, so
|S|·2^(−3860) ≤ Σ_{s ∈ V^2018} ∏ᵢ (1/2)^(sᵢ) = (Σ_{v∈V} (1/2)^v)^2018 = (1009/1024)^2018
(`Finset.prod_univ_sum`), i.e. |S| ≤ 2^3860·(2018/2048)^2018.  (Never let `norm_num` see
the big powers.)
-/
theorem putnam_2018_b6
  (S : Finset (Fin 2018 → ℤ))
  (hS : S = {s : Fin 2018 → ℤ | (∀ i : Fin 2018, s i ∈ ({1, 2, 3, 4, 5, 6, 10} : Set ℤ)) ∧ (∑ i : Fin 2018, s i) = 3860}) :
  S.card ≤ 2 ^ 3860 * ((2018 : ℝ) / 2048) ^ 2018 := by
  classical
  have hmem : ∀ s, s ∈ S ↔ (∀ i : Fin 2018, s i ∈ ({1, 2, 3, 4, 5, 6, 10} : Set ℤ)) ∧
      (∑ i : Fin 2018, s i) = 3860 := fun s => by rw [← Finset.mem_coe, hS]; rfl
  let V : Finset ℤ := {1, 2, 3, 4, 5, 6, 10}
  let f : ℤ → ℝ := fun v => (1 / 2 : ℝ) ^ v.toNat
  have hf0 : ∀ v, 0 ≤ f v := fun v => by positivity
  -- each s ∈ S has weight (1/2)^3860
  have hw : ∀ s ∈ S, ∏ i, f (s i) = (1 / 2 : ℝ) ^ 3860 := by
    intro s hs
    obtain ⟨hv, hsum⟩ := (hmem s).mp hs
    have hpos : ∀ i, 0 ≤ s i := by
      intro i; have := hv i; simp only [Set.mem_insert_iff, Set.mem_singleton_iff] at this; omega
    have hN0 : ((∑ i, (s i).toNat : ℕ) : ℤ) = 3860 := by
      push_cast; rw [← hsum]; exact Finset.sum_congr rfl fun i _ => Int.toNat_of_nonneg (hpos i)
    have hN : (∑ i, (s i).toNat) = 3860 := by exact_mod_cast hN0
    simp only [f]
    rw [Finset.prod_pow_eq_pow_sum, hN]
  have hsub : S ⊆ Fintype.piFinset (fun _ : Fin 2018 => V) := by
    intro s hs
    rw [Fintype.mem_piFinset]
    intro i
    have := ((hmem s).mp hs).1 i
    simp only [Set.mem_insert_iff, Set.mem_singleton_iff] at this
    simp only [V, Finset.mem_insert, Finset.mem_singleton]
    exact this
  have hV : ∑ v ∈ V, f v = 1009 / 1024 := by
    simp only [V, f]
    rw [Finset.sum_insert (by decide), Finset.sum_insert (by decide), Finset.sum_insert (by decide),
      Finset.sum_insert (by decide), Finset.sum_insert (by decide), Finset.sum_insert (by decide),
      Finset.sum_singleton]
    have t1 : Int.toNat 1 = 1 := rfl
    have t2 : Int.toNat 2 = 2 := rfl
    have t3 : Int.toNat 3 = 3 := rfl
    have t4 : Int.toNat 4 = 4 := rfl
    have t5 : Int.toNat 5 = 5 := rfl
    have t6 : Int.toNat 6 = 6 := rfl
    have t10 : Int.toNat 10 = 10 := rfl
    rw [t1, t2, t3, t4, t5, t6, t10]
    norm_num
  have key : (S.card : ℝ) * (1 / 2 : ℝ) ^ 3860 ≤ (1009 / 1024 : ℝ) ^ 2018 := by
    calc (S.card : ℝ) * (1 / 2 : ℝ) ^ 3860 = ∑ s ∈ S, ∏ i, f (s i) := by
          rw [Finset.sum_congr rfl hw, Finset.sum_const, nsmul_eq_mul]
      _ ≤ ∑ s ∈ Fintype.piFinset (fun _ : Fin 2018 => V), ∏ i, f (s i) :=
          Finset.sum_le_sum_of_subset_of_nonneg hsub
            (fun s _ _ => Finset.prod_nonneg fun i _ => hf0 _)
      _ = ∏ _i : Fin 2018, ∑ v ∈ V, f v := (Finset.prod_univ_sum _ _).symm
      _ = (1009 / 1024 : ℝ) ^ 2018 := by rw [hV, Finset.prod_const, Finset.card_univ,
          Fintype.card_fin]
  rw [show (2018 : ℝ) / 2048 = 1009 / 1024 by norm_num]
  have h1 : (2 : ℝ) ^ 3860 * (1 / 2 : ℝ) ^ 3860 = 1 := by rw [← mul_pow]; norm_num
  have h2 : (0 : ℝ) ≤ 2 ^ 3860 := by positivity
  calc (S.card : ℝ) = 2 ^ 3860 * ((S.card : ℝ) * (1 / 2 : ℝ) ^ 3860) := by
        rw [mul_comm (S.card : ℝ), ← mul_assoc, h1, one_mul]
    _ ≤ 2 ^ 3860 * (1009 / 1024 : ℝ) ^ 2018 := mul_le_mul_of_nonneg_left key h2
