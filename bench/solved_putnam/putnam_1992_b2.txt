import Mathlib

open Topology Filter Nat Function Polynomial

/-!
Putnam 1992 B2.  `(1 + X + X² + X³)ⁿ = (1 + X²)ⁿ (1 + X)ⁿ = ∑ⱼ C(n,j) X^{2j} (1 + X)ⁿ`, so the
coefficient of `X^k` is `∑ⱼ C(n,j) [2j ≤ k] C(n, k - 2j)`.  The sum over `j ≤ n` equals the
sum over `j ≤ k`: both agree with the sum over `j < n + k + 1` (extra terms vanish because
`C(n,j) = 0` for `j > n`, resp. `2j > k` for `j > k`).
-/
theorem putnam_1992_b2
    (Q : ℕ → ℕ → ℕ)
    (hQ : Q = fun n k ↦ coeff ((1 + X + X ^ 2 + X ^ 3) ^ n) k)
    (n k : ℕ) :
    Q n k = ∑ j ∈ Finset.Iic k, choose n j * (if 2 * j ≤ k then choose n (k - 2 * j) else 0) := by
  subst hQ
  simp only
  have hfac : (1 + X + X ^ 2 + X ^ 3 : ℕ[X]) = (1 + X ^ 2) * (1 + X) := by ring
  have hexp : (1 + X ^ 2 : ℕ[X]) ^ n = ∑ j ∈ Finset.range (n + 1), C (n.choose j) * X ^ (2 * j) := by
    rw [add_comm, add_pow]
    refine Finset.sum_congr rfl fun j _ => ?_
    rw [one_pow, mul_one, ← pow_mul, mul_comm (X ^ _), ← map_natCast C, mul_comm 2 j, Nat.cast_id]
  rw [hfac, mul_pow, hexp, Finset.sum_mul, Polynomial.finset_sum_coeff]
  simp_rw [mul_assoc, Polynomial.coeff_C_mul, Polynomial.coeff_X_pow_mul', Polynomial.coeff_one_add_X_pow, Nat.cast_id]
  set F : ℕ → ℕ := fun j => n.choose j * (if 2 * j ≤ k then n.choose (k - 2 * j) else 0) with hF
  have h1 : ∑ j ∈ Finset.range (n + 1), F j = ∑ j ∈ Finset.range (n + k + 1), F j := by
    apply Finset.sum_subset
    · intro j; simp only [Finset.mem_range]; omega
    · intro j _ hj
      simp only [Finset.mem_range, not_lt] at hj
      simp [hF, Nat.choose_eq_zero_of_lt (by omega : n < j)]
  have h2 : ∑ j ∈ Finset.Iic k, F j = ∑ j ∈ Finset.range (n + k + 1), F j := by
    apply Finset.sum_subset
    · intro j; simp only [Finset.mem_Iic, Finset.mem_range]; omega
    · intro j _ hj
      simp only [Finset.mem_Iic, not_le] at hj
      simp [hF, show ¬ (2 * j ≤ k) by omega]
  have hL : ∑ j ∈ Finset.range (n + 1),
      n.choose j * (if 2 * j ≤ k then n.choose (k - 2 * j) else 0) = ∑ j ∈ Finset.range (n + 1), F j := rfl
  rw [hL, h1, ← h2]

#print axioms putnam_1992_b2
